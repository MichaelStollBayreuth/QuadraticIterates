\chapter{Auxiliary theory}

The paper carries out several arguments by hand that are really instances of general theory:
degrees of multiquadratic extensions, invariants of a $2$-group acting on an $\FF_2$-vector
space, abelian quotients of iterated wreath products, and Möbius quotients of a strong
divisibility sequence. In the formalization these are developed separately, in the generality
they naturally have, under \texttt{QuadraticIterates/Mathlib/}; they are candidates for
upstreaming into Mathlib. This chapter records them, since the proofs of Chapters~1--3 refer to
them.

\section{Multiquadratic extensions}

\begin{lemma}[Degree of a multiquadratic extension]
  \label{lem:multiquadratic}
  \lean{multiquadraticRelations, mem_multiquadraticRelations, multiquadratic_degree,
    multiquadratic_degree_family, IntermediateField.adjoin_insert,
    IntermediateField.finrank_adjoin_insert,
    IntermediateField.finrank_adjoin_insert_of_not_isSquare,
    IntermediateField.finrank_adjoin_sqrt_le, IntermediateField.finrank_adjoin_finset_sqrt_le,
    IntermediateField.finrank_adjoin_sqrt_le_two_pow_ncard,
    IntermediateField.finrank_adjoin_sqrt_eq, multiquadraticRelations_finrank_le,
    multiquadraticRelations_insert_finrank}
  \leanok
  Let $F$ be a field of characteristic $\ne 2$ and $(x_i)_{i \in s}$ a finite family of square
  roots of elements $r_i \in F^*$ inside an extension. Then
  $[F(x_i : i \in s) : F] = 2^{\#s - \dim_{\FF_2} R(s)}$, where $R(s) \subseteq \FF_2^{s}$ is the
  space of \emph{relations}: the $\varepsilon$ with $\prod_{i : \varepsilon_i = 1} r_i$ a square in
  $F$. (The family need not be injective: a repeated root counts once in $\#s$ and once in
  $R(s)$.) In particular $[F(x_i) : F] \le 2^{\#s}$, with equality iff there are no relations; and
  adjoining one further square root either doubles the degree or leaves it unchanged.
\end{lemma}
\begin{proof}
  \leanok
  Induction on $\#s$, the step being the descent Lemma~\ref{lem:descent}: the degree doubles
  exactly when the new radicand is not a square in the field generated so far.
\end{proof}

\begin{lemma}[Square descent]
  \label{lem:descent}
  \uses{def:twoindep}
  \lean{IntermediateField.square_descent,
    IntermediateField.exists_sq_eq_algebraMap_of_isSquare_mul_prod,
    IntermediateField.square_descent_step, IntermediateField.isSquare_algebraMap_iff,
    IntermediateField.not_isSquare_algebraMap_of_sqrt_notMem,
    IntermediateField.isSquare_algebraMap_bot_iff, IntermediateField.mem_adjoin_sqrt_iff,
    sqClass_mul, sqClass_prod, sqClass_prod_zpow, sqClass_eq_zero_iff,
    isSquare_prod_iff_sum_sqClass_eq_zero}
  \leanok
  An element of $F^*$ is a square in a multiquadratic extension $F(\sqrt{c_y} : y \in s)$ if and
  only if it becomes a square in $F$ after multiplication by a subproduct of the $c_y$.
\end{lemma}
\begin{proof}
  \leanok
  Adjoining one square root at a time; in each step, an element of the smaller field whose square
  root lies in the quadratic extension differs from a square by the radicand.
\end{proof}

\begin{lemma}[Relations among the shifted roots]
  \label{lem:relations}
  \uses{lem:multiquadratic}
  \lean{rootRelations, mem_rootRelations, rootRelations_eq_bot_iff, rootRelations_finrank_le,
    multiquadraticRelations_univ, multiquadraticRelations_finrank_eq_rootRelations,
    all_ones_mem_rootRelations, Function.ExtendByZero.linearMap_injective,
    exists_mem_multiquadraticRelations_insert_iff, multiquadraticRelations_insert_inf_ker_proj,
    indicator_mem_multiquadraticRelations_iff}
  \leanok
  For a family $(r_i)_{i \in \iota}$ of nonzero elements of $F$, the relation space
  $R \subseteq \FF_2^{\iota}$ of the full family is the space of relations of
  Lemma~\ref{lem:multiquadratic} for $s = \iota$; it vanishes iff the classes of the $r_i$ in
  $F^*/F^{*2}$ are linearly independent, and $\dim R \le \#\iota$. The all-ones vector lies in
  $R$ if and only if $\prod_i r_i$ is a square in $F$.
\end{lemma}
\begin{proof}
  \uses{def:twoindep, lem:descent}
  \leanok
  Unfolding the definition of the relation space.
\end{proof}

\begin{lemma}[Invariants of a $p$-group]
  \label{lem:invariant}
  \uses{lem:relations}
  \lean{IsPGroup.exists_ne_zero_mem_fixedPoints_of_smul_mem, invariant_submodule_all_ones,
    rootRelations_invariant, rootRelations_all_ones}
  \leanok
  Let a $p$-group $G$ act on an $\FF_p$-vector space $M$, and let $V \subseteq M$ be a finite
  nonzero $G$-stable subspace. Then $V^G \ne 0$. If moreover $M = \FF_p^{\iota}$ and $G$ acts by
  permuting the coordinates \emph{transitively}, then the all-ones vector lies in $V$.
  (We use this with $p = 2$.)
\end{lemma}
\begin{proof}
  \leanok
  For the first statement, $\#V$ is a power of $p$, and $\#V \equiv \#V^G \bmod p$ by Mathlib's
  \lname{IsPGroup.card\_modEq\_card\_fixedPoints}; since $0 \in V^G$, this gives $\#V^G > 1$
  (Mathlib's \lname{IsPGroup.exists\_fixed\_point\_of\_prime\_dvd\_card\_of\_fixed\_point}).
  For the second, a nonzero invariant vector has constant coordinates along the orbits of $G$ on
  $\iota$, and there is only one orbit; scaling makes the constant equal to $1$.
\end{proof}

\section{Wreath products}

\begin{lemma}[Sylow $2$-subgroups of symmetric groups]
  \label{lem:sylowwreath}
  \lean{Sylow.mulEquivIteratedWreathProduct, IteratedWreathProduct.card,
    IsPGroup.exists_injective_monoidHom_iteratedWreathProduct,
    MonoidHom.nonempty_mulEquiv_iff_card_eq}
  \leanok
  A Sylow $2$-subgroup of the symmetric group on $2^n$ letters is isomorphic to $\Ct{n}$, which
  has order $2^{2^n - 1}$; hence a $2$-group of permutations of at most $2^n$ points embeds into
  $\Ct{n}$. Moreover, an injective homomorphism between finite groups of equal order is an
  isomorphism, which is how ``$\Om_n \cong \Ct{n}$'' is obtained from an embedding plus a degree
  count.
\end{lemma}
\begin{proof}
  \leanok
  The first statement is in Mathlib (\texttt{Sylow.mulEquivIteratedWreathProduct}); the second
  follows by extending the action to $2^n$ points and passing to a Sylow subgroup, and the last
  one is elementary.
\end{proof}

\begin{lemma}[The abelian quotient of $\Ct{n}$]
  \label{lem:wreathab}
  \lean{wreath_max_elem_ab, IteratedWreathProduct.card_hom, RegularWreathProduct.card_hom,
    RegularWreathProduct.homEquiv, Nat.card_monoidHom_multiplicative_zmod_two,
    Nat.card_monoidHom_multiplicative_zmod, Nat.card_addMonoidHom_zmod,
    IntermediateField.isGalois_adjoin_of_sq_eq_algebraMap,
    IntermediateField.algEquiv_adjoin_sq_eq_one, AlgEquiv.apply_eq_or_eq_neg_of_sq_eq_algebraMap}
  \leanok
  $\#\Hom(\Ct{n}, C_2) = 2^n$; equivalently, the largest elementary abelian $2$-quotient of
  $\Ct{n}$ is $C_2^n$. A field extension generated by square roots of base-field elements is
  Galois with elementary abelian group of exponent $2$, so its degree equals the number of
  homomorphisms of its Galois group to $C_2$.
\end{lemma}
\begin{proof}
  \leanok
  For a regular wreath product $D \wr Q$ and an abelian group $A$, a homomorphism $D \wr Q \to A$
  is the same as a pair of homomorphisms $D \to A$ and $Q \to A$ (on the base group $D^Q$ it
  takes the same values on all coordinates, since these are conjugate under $Q$), giving
  $\#\Hom(D \wr Q, A) = \#\Hom(D, A) \cdot \#\Hom(Q, A)$; induction on $n$ then gives
  $\#\Hom(\Ct{n}, C_2) = \#\Hom(C_2, C_2)^n = 2^n$. The statement about elementary abelian
  groups is the identity $\#G = \#\Hom(G, C_p)$ for a finite $\FF_p$-vector space $G$, i.e.\
  $\dim G^{\vee} = \dim G$ (for $p = 2$, commutativity follows from the exponent).
\end{proof}

\section{Möbius quotients of a strong divisibility sequence}

\begin{lemma}[Strong divisibility from a translation congruence]
  \label{lem:strongdvd}
  \lean{associated_gcd_of_dvd_sub, gcd_eq_normalize_of_dvd_sub}
  \leanok
  Let $(c_n)$ be a sequence in a GCD domain with $c_0 = 0$ and $c_m \dvd c_{m+j} - c_j$ for all
  $m, j$. Then $\gcd(c_m, c_n)$ is associated to $c_{\gcd(m,n)}$.
\end{lemma}
\begin{proof}
  \leanok
  The Euclidean algorithm on the indices: the hypothesis gives
  $\gcd(c_{m+j}, c_j) = \gcd(c_m, c_j)$, which is the step of the subtractive algorithm; the
  junk value $c_0 = 0$ makes the base case $\gcd(c_m, c_0) = c_m$ come out right.
\end{proof}

\begin{lemma}[Integrality of the Möbius quotients]
  \label{lem:moebiusfactor}
  \lean{moebiusFactorK, moebiusFactorK_eq_div, moebiusFactorK_eq_prod, moebiusFactorR, numProd,
    denProd, denProd_dvd_numProd, moebiusFactorK_isInteger, algebraMap_moebiusFactorR,
    prod_moebiusFactorR, moebiusFactorR_ne_zero, moebiusFactorR_mul_denProd,
    factorization_gcd_min, UniqueFactorizationMonoid.normalizedFactors_gcd, isInteger_div_iff_dvd}
  \leanok
  Let $R$ be a UFD and $(c_n)$ a sequence in $R \setminus \{0\}$ (on positive indices) which is a
  strong divisibility sequence. Then each Möbius quotient
  $b_n := \prod_{d \dvd n} c_d^{\mu(n/d)}$ lies in $R$ and is nonzero, and Möbius inversion
  holds: $c_n = \prod_{d \dvd n} b_d$.
\end{lemma}
\begin{proof}
  \uses{lem:moebiussign}
  \leanok
  Write the quotient as $N/D$ with $N$ the product of the $c_d$ with $\mu(n/d) = 1$ and $D$ the
  product of those with $\mu(n/d) = -1$. At each prime $p$, strong divisibility bounds
  $v_p(c_{\gcd(d,d')})$ below by $\min(v_p(c_d), v_p(c_{d'}))$, and a counting argument on the
  divisor lattice shows $v_p(D) \le v_p(N)$; hence $D \dvd N$.
\end{proof}

\begin{theorem}[Constant valuation shape]
  \label{thm:valshape}
  \uses{def:gammaseq, lem:moebiusfactor}
  \lean{factorization_periodic_shape, factorization_moebiusFactorR_shape,
    moebiusFactorR_isRelPrime, factorization_eq_of_dvd_sub, one_le_factorization_iff_dvd,
    factorization_eq_zero_iff_not_dvd, pow_dvd_iff_le_factorization,
    emultiplicity_eq_factorization,
    QuadraticIterates.factorization_gammaSeq_shape,
    QuadraticIterates.factorization_gammaSeq_of_dvd_eval_zero,
    QuadraticIterates.gammaSeq_period, QuadraticIterates.sq_dvd_gammaSeq_succ_sub,
    QuadraticIterates.pow_succ_dvd_gammaSeq_succ_sub,
    QuadraticIterates.gammaSeq_period_of_dvd_succ_sub_eval_zero,
    QuadraticIterates.gammaSeq_one_associated_eval_zero, Associated.factorization_eq}
  \leanok
  For the iteration sequence of an even polynomial with $\varepsilon^2 = 1$ over a UFD, and each
  normalized prime $p$, there are $m \ge 1$ and $E$ with $v_p(\gamma_n) = E$ for $m \dvd n$ and
  $v_p(\gamma_n) = 0$ otherwise. Consequently the valuation of each Möbius quotient $\beta_n$ is
  supported on the single index $n = m$, so the $\beta_n$ are pairwise coprime.
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:moebiussign}
  Two cases. If $p \dvd g(0)$, the valuation is constantly $v_p(g(0))$, since
  $\gamma_n^2 \dvd \gamma_{n+1} - g(0)$ forces $v_p(\gamma_{n+1}) = v_p(g(0))$ by induction. If
  not, let $m$ be minimal with $p \dvd \gamma_m$ and $E := v_p(\gamma_m)$; then
  $p^{E+1} \dvd \gamma_{m+1} - g(0)$, whence $p \nmid \gamma_{m+1}$, and the sequence is periodic
  mod $p^{E+1}$ with period $m$ from index $2$ on --- this is the periodicity argument of the
  proof of Lemma~\ref{lem:1.1}~b), isolated and stated for an arbitrary sequence with these
  properties (\lname{factorization\_periodic\_shape}).
\end{proof}

\begin{lemma}[Möbius sums]
  \label{lem:moebiussign}
  \lean{moebius_sign_partition, prod_pow_moebius_eq_div, moebius_antidiag_sum_zero,
    beta_radical, moebius_restricted_sum, indicator_moebius_nonneg, moebius_transform_nonneg,
    ArithmeticFunction.sum_divisors_moebius}
  \leanok
  For squarefree $n' > 1$, the divisors of $n'$ split into two sets of equal cardinality
  according to the sign of $\mu$, so a Möbius product $\prod_{d \dvd n} x_d^{\mu(n/d)}$ is a
  quotient of two products with the same number of factors; and
  $\sum_{d \dvd n} \mu(n/d) = 0$ for $n \ge 2$.
\end{lemma}
\begin{proof}
  \leanok
  Standard properties of $\mu$. The equal-cardinality statement is the binomial identity
  \[ \sum_{k} (-1)^k \binom{r}{k} = 0, \]
  where $r \ge 1$ is the number of prime divisors of $n'$.
\end{proof}

\section{Squares}

\begin{lemma}[Non-squares mod $m$]
  \label{lem:zmodsquare}
  \lean{ZMod.isSquare_neg_one_of_isSquare_div, ZMod.not_isSquare_neg_one_of_dvd,
    ZMod.not_isSquare_neg_one_of_emod_four_eq_three, ZMod.not_isSquare_neg_one_of_emod_eight_eq_six,
    ZMod.not_isSquare_neg_one_of_four_dvd, ZMod.isUnit_intCast_of_isCoprime_of_dvd_add,
    ZMod.intCast_eq_neg_intCast_of_dvd_add}
  \leanok
  If $P \equiv -Q \bmod m$ with $Q$ a unit mod $m$, and $P/Q$ is a square in $\QQ$, then $-1$ is
  a square mod $m$. Moreover $-1$ is not a square mod $m$ whenever $m$ has a divisor
  $\equiv 3 \bmod 4$, and not mod $4$.
\end{lemma}
\begin{proof}
  \leanok
  Reduce a representation $P/Q = (u/v)^2$ modulo $m$ and use that $Q$ is invertible; the
  supplementary statements are the standard obstructions to $-1$ being a square.
\end{proof}

\begin{lemma}[Squares and coprimality]
  \label{lem:coprimesquare}
  \lean{Int.isSquare_abs_of_isSquare_prod_of_pairwise_isCoprime, Finset.prod_zpow_eq_zpow_sum₀}
  \leanok
  If a product of pairwise coprime integers is a square, then the absolute value of each factor
  is a square.
\end{lemma}
\begin{proof}
  \leanok
  Compare the $p$-adic valuations: each prime occurs in exactly one factor.
\end{proof}

\section{Even polynomials}

\begin{lemma}[Even polynomials are polynomials in $X^2$]
  \label{lem:evenpoly}
  \lean{Polynomial.even_eq_comp_X_sq_add_C, Polynomial.eq_expand_two_contract_of_comp_neg_X,
    Polynomial.expand_two_comp_neg_X, Polynomial.X_sq_add_C_comp_neg_X,
    Polynomial.comp_neg_X_eq_or_eq_neg_of_associated, Polynomial.natDegree_comp_neg_X,
    Polynomial.expand_contract_of_forall_coeff_eq_zero,
    Polynomial.coeff_eq_zero_of_comp_neg_X_eq_of_not_two_dvd, Polynomial.comp_dvd_comp_iff}
  \leanok
  Over a domain of characteristic $\ne 2$, a polynomial fixed by $X \mapsto -X$ is a polynomial
  in $X^2$, hence in $X^2 + c$ for any $c$. A polynomial merely \emph{associated} to its
  reflection satisfies $p(-X) = p$ or $p(-X) = -p$. Over a field, $p \circ r \dvd q \circ r$
  implies $p \dvd q$ for every nonconstant $r$.
\end{lemma}
\begin{proof}
  \leanok
  The odd coefficients vanish because $2 \ne 0$; for the second statement, the unit relating $p$
  to $p(-X)$ is a constant squaring to $1$. For the last statement, the remainder of $q$ modulo
  $p$, composed with $r$, is divisible by $p \circ r$ but has smaller degree, so it vanishes.
\end{proof}

\begin{lemma}[Pairing the irreducible factors]
  \label{lem:involution}
  \lean{Multiset.exists_add_map_of_involutive, normalizedFactors_map_mulEquiv_eq,
    normalize_map_normalize, exists_normalizedFactors_eq_add_map}
  \leanok
  A fixed-point-free involution of a multiset splits it as $N + \tau(N)$. Consequently, if
  $\sigma$ is an involutive automorphism of a UFD and $p$ is a reducible element with
  $\sigma(p) \sim p$ whose $\sigma$-invariant divisors (up to associates) are all trivial, then
  the normalized factors of $p$ pair up under $\mathrm{normalize} \circ \sigma$. For
  $\sigma = (X \mapsto -X)$ on the polynomial ring over a field, this yields the factorization
  $\pm p = g \cdot g(-X)$ of an even polynomial without nontrivial even divisors.
\end{lemma}
\begin{proof}
  \leanok
  Induction on the multiset, removing an orbit $\{q, \tau q\}$ at a time; that
  $\mathrm{normalize} \circ \sigma$ preserves the multiset of normalized factors is
  functoriality of factorization under $\sigma$, and it has no fixed point there because a fixed
  normalized factor would be a nontrivial $\sigma$-invariant divisor.
\end{proof}
